cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(InterpFixTests LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "InterpFix tests require Windows MSVC x86 (-A Win32).")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(METAHOOK_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../MetaHook" CACHE PATH "MetaHook SDK source tree")
enable_testing()
add_executable(interpfix_interpolation_test interpolation_test.cpp ../src/interpolation.cpp)
include(../cmake/SDK.cmake)
interpfix_use_sdk(interpfix_interpolation_test)
add_executable(interpfix_hooks_test hooks_test.cpp ../src/privatehook.cpp ../src/interpolation.cpp)
interpfix_use_sdk(interpfix_hooks_test)
add_test(NAME interpfix_interpolation COMMAND interpfix_interpolation_test)
add_test(NAME interpfix_hooks COMMAND interpfix_hooks_test)
set_tests_properties(interpfix_interpolation interpfix_hooks PROPERTIES TIMEOUT 30)
